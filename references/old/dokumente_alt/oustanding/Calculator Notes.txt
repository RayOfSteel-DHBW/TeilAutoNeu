Private Sub tagab_AfterUpdate()
On Error Resume Next
a = Me!tagab
B = InStr(1, a, ".")
If B = 0 Then
    Me!tagab = DateValue(Me!tagab & "." & Me!Monat & "." & Me!jahr)
    Me!tagab = Format(Me!tagab, "dd.mm.yyyy")
Else
    Me!tagab = DateValue(Me!tagab)
End If
If DateValue(Me!tagab) < Date - 365 Then
    MsgBox "Mit dem Datum stimmt was nicht!" & Chr(13) & "Sie haben den Wert '" & a & "' eingegeben."
    Me!tagab = Null
    SendKeys "+{tab}"
    Exit Sub
End If
Me!tagab = CLng(DateSerial(Mid(Me!tagab, 7, 4), Mid(Me!tagab, 4, 2), Mid(Me!tagab, 1, 2)))
Me!tagab = Format(Me!tagab, "dd.mm.yyyy")
Me!tagbis = Me!tagab
End Sub

End Sub

Private Sub Zeitab_AfterUpdate()
'On Error Resume Next

Select Case Right(Me!Zeitab, 1)
    Case "0" To "9"
     Select Case Me!Zeitab
        Case Is > 23
            MsgBox "Mit der Uhrzeit stimmt was nicht!"
            SendKeys "+{tab}"
            Exit Sub
     End Select
    B = 0
    Case ","
    a = Len(Me!Zeitab)
    B = 30
    Me!Zeitab = Left(Me!Zeitab, a - 1)
    Case Else
    MsgBox "Mit der Uhrzeit stimmt was nicht!"
    SendKeys "+{tab}"
    Exit Sub
End Select



Me!Zeitab = TimeValue(Me!Zeitab & ":" & B)
Me!Zeitab = Format(Me!Zeitab, "hh:nn")

End Sub

Private Sub zeitbis_AfterUpdate()
On Error GoTo zeiterror
Select Case Right(Me!zeitbis, 1)
    Case "0" To "9"
    B = 0
    Case ","
    a = Len(Me!zeitbis)
    B = 30
    Me!zeitbis = Left(Me!zeitbis, a - 1)
    Case Else
    MsgBox "Mit der Uhrzeit stimmt was nicht!"
    SendKeys "+{tab}"
    Exit Sub
End Select
If Me!zeitbis = 24 Then
    Me!tagbis = CLng(DateSerial(Mid(Me!tagbis, 7, 4), Mid(Me!tagbis, 4, 2), Mid(Me!tagbis, 1, 2)))
    Me!tagbis = Me!tagbis + 1
    Me!tagbis = Format(Me!tagbis, "dd.mm.yyyy")
    Me!zeitbis = 0
End If

Me!zeitbis = TimeValue(Me!zeitbis & ":" & B)
Me!zeitbis = Format(Me!zeitbis, "hh:nn")

zeitende:
Exit Sub

zeiterror:
MsgBox "Bitte Uhrzeit erst löschen und dann neu eingeben!"
SendKeys "+{tab}"
Resume zeitende
End Sub


Private Sub berechnen_neu()
Dim datab, datbis, abzug As Double
Dim std_zählen As Integer
datab = CDbl(DateValue(Me!tagab) + TimeValue(Me!Zeitab))
datbis = CDbl(DateValue(Me!tagbis) + TimeValue(Me!zeitbis))

std_zählen = 0
std_unter_über = 0

Me!kmgeld_neu = 0

Me!zeitgeld_unter8 = 0
Me!zeitgeld_über8 = 0
Me!zeitgeld_nacht = 0

Me!Summe = 0
Me!std_über = 0
Me!std_unter = 0
Me!std_nacht = 0

If DLookup("kmtarif2", "global") = True And DLookup("kmtarif3", "global") = True Then
    If Me!gefahren > DLookup("kilometer3", "global") Then
        Me!über500 = Me!gefahren - DLookup("kilometer3", "global")
        Me!über100 = DLookup("kilometer3", "global") - DLookup("kilometer", "global")
        Me!gefahren = DLookup("kilometer", "global")
        Me!km100geld_neu = Me!über100 * Me!km100_neu
        Me!km500geld_neu = Me!über500 * Me!km500_neu
    ElseIf Me!gefahren > DLookup("kilometer", "global") Then
        Me!über100 = Me!gefahren - DLookup("kilometer", "global")
        Me!gefahren = DLookup("kilometer", "global")
        Me!km100geld_neu = Me!über100 * Me!km100_neu
    End If
ElseIf DLookup("kmtarif2", "global") = True Then
    If Me!gefahren > DLookup("kilometer", "global") Then
        Me!über100 = Me!gefahren - DLookup("kilometer", "global")
        Me!gefahren = DLookup("kilometer", "global")
        Me!km100geld_neu = Me!über100 * Me!km100_neu
    End If
End If

Me!kmgeld_neu = Me!gefahren * Me!Kilometertarif_neu

If globgebühr = True Then
    Me!gebühren = Me!gebühren + globgebdm
End If

Do Until Year(datab) = Year(datbis) And Day(datab) = Day(datbis) And Hour(datab) = Hour(datbis) And minute(datab) = minute(datbis)
    datab = datab + halbe_std
    std_zählen = std_zählen + 0.5
    Select Case Hour(datab) & Format(minute(datab), "00")
        Case 0 To 800, 2330
            std_nacht = std_nacht + 0.5
        Case Else
            std_unter_über = std_unter_über + 0.5
            If std_unter_über <= globlangtarif Then
                Me!std_unter = Me!std_unter + 0.5
            Else
                Me!std_über = Me!std_über + 0.5
            End If
    End Select
Loop

Me!zeitgeld_unter8 = Me!zeittarif_unter8 * Me!std_unter
Me!zeitgeld_über8 = Me!zeittarif_über8 * Me!std_über
Me!zeitgeld_nacht = Me!zeittarif_nacht * Me!std_nacht

Me!Summe = Me!kmgeld_neu + Me!km100geld_neu + Me!km500geld_neu + Me!zeitgeld_unter8 + Me!zeitgeld_über8 + Me!zeitgeld_nacht + Me!gebühren
End Sub
